<time.h>

No cabeçalho <time.h> estão localizadas as funções, variáveis e macros para manipulação de unidades de tempo, como ano, mês, dia, hora, minutos e segundos. A partir das funções fornecidas é possível obter a data e o horário do sistema, calcular intervalos de tempo, entre outras funcionalidades.

É um cabeçalho muito poderoso para o controle da execução do programa e para a manipulação de ações que têm datas e horas como referência.

Variáveis e Tipos

  • clock_t: Variável que tem por finalidade armazenar valores relacionados a pulsos de clock do processador, como o valor fornecido pela função clock().
  • time_t: Variável que armazena os valores numéricos brutos relacionados à data e hora do sistema (frequentemente em segundos desde a Epoch). Várias funções recebem ou retornam este tipo.
  • size_t: Variável utilizada para representar o tamanho que uma determinada variável e/ou tipo ocupa na memória.
  • struct tm: Estrutura de dados que armazena valores de calendário (data e hora desmembrados), contendo 9 variáveis inteiras (segundos, minutos, horas, dia do mês, mês, ano, dia da semana, dia do ano e horário de verão).

Macros

  • CLOCKS_PER_SEC: Macro responsável por armazenar a quantidade de pulsos por segundo do processador da máquina na qual o programa está sendo executado. É usada para converter o retorno de clock() em segundos.
  • NULL: Ponteiro utilizado para indicar que uma referência não aponta para posição de memória alguma.

Funções

clock() time() difftime() mktime() asctime() ctime() gmtime() localtime() strftime()

long int clock(void)

Retorna o tempo de processamento consumido pelo programa.

Recebe: void (não recebe parâmetros).
Retorna: A quantidade de pulsos de clock (tipo clock_t ou long int) desde a inicialização do programa. Divida por CLOCKS_PER_SEC para obter o tempo em segundos.

Exemplo

        
// EXEMPLO CLOCK()
#include <stdio.h>
#include <time.h>

int main() { 
    printf("Pulsos de clock até o momento: %ld\n", clock()); 
    printf("printf Aleatório\n"); 
    printf("Mais um teste da quantidade de pulsos de clock: %ld\n", clock());

    return 0; 
}

time_t time(time_t *ponteiro_tempo)

Retorna a data e hora atuais do sistema (geralmente como o número de segundos desde 01/01/1970).

Recebe: Um ponteiro para uma variável do tipo time_t (opcional, pode ser NULL).
Retorna: O valor do tempo atual no tipo time_t, e também o atribui ao ponteiro se ele não for NULL.

Exemplo

        
// EXEMPLO TIME()
#include <stdio.h>
#include <time.h>

int main() { 
    // Passando NULL para retornar apenas o valor
    time_t tempo = time(NULL); 
    printf("%s", ctime(&tempo));

    // Passando o endereço para atribuir o valor diretamente na variável
    time(&tempo); 
    printf("Outro teste: %s", ctime(&tempo));

    return 0; 
}

double difftime(time_t tempo, time_t tempo0)

Calcula a diferença, em segundos, entre dois valores de tempo.

Recebe: Duas variáveis do tipo time_t.
Retorna: A diferença (tempo - tempo0) em segundos como um double.

Exemplo

        
// EXEMPLO DIFFTIME()
#include <stdio.h>
#include <time.h>

int main() { 
    struct tm tempo = {0}, tempo0 = {0};

    // Tempo 1 (84 segundos)
    tempo.tm_sec = 2 * 42; 
    tempo.tm_min = 1; tempo.tm_hour = 1; 
    tempo.tm_mday = 1; tempo.tm_mon = 1; tempo.tm_year = 1; tempo.tm_wday = 1;

    // Tempo 0 (42 segundos)
    tempo0.tm_sec = 42; 
    tempo0.tm_min = 1; tempo0.tm_hour = 1; 
    tempo0.tm_mday = 1; tempo0.tm_mon = 1; tempo0.tm_year = 1; tempo0.tm_wday = 1;

    printf("Diferença: %lf\n", difftime(mktime(&tempo), mktime(&tempo0)));

    return 0; 
}

time_t mktime(struct tm *tempo)

Converte uma estrutura de tempo desmembrada (struct tm) no formato de calendário para um valor simples de time_t.

Recebe: Um ponteiro para uma variável struct tm.
Retorna: O valor do tipo time_t correspondente e ajusta automaticamente os valores fora do limite na estrutura original.

Exemplo

        
// EXEMPLO MKTIME()
#include <stdio.h>
#include <time.h>

int main() { 
    struct tm tempo = {0};

    tempo.tm_sec = 42; 
    tempo.tm_min = 59; 
    tempo.tm_hour = 23; 
    tempo.tm_mday = 25; 
    tempo.tm_mon = 12; // Dezembro (0-11, então 12 vira Jan do próximo ano)
    tempo.tm_year = 10191 - 1900; 

    time_t tempoConvertido = mktime(&tempo);
    printf("%s", ctime(&tempoConvertido));

    return 0; 
}

char *asctime(const struct tm *tp)

Converte uma estrutura struct tm para uma string legível formatada.

Recebe: Um ponteiro para struct tm contendo os dados de tempo.
Retorna: Uma string com a formatação: 'DiaSem Mês Dia Hora:Min:Seg Ano\n\0'.

Exemplo

        
// EXEMPLO ASCTIME()
#include <stdio.h>
#include <time.h>

int main() { 
    struct tm tempo = {0};
    
    tempo.tm_sec = 42; 
    tempo.tm_min = 59; 
    tempo.tm_hour = 23; 
    tempo.tm_mday = 25; 
    tempo.tm_mon = 12 - 1; // Dezembro é 11
    tempo.tm_year = 10191 - 1900; 
    tempo.tm_wday = 3; // Quarta-feira

    printf("%s", asctime(&tempo));

    return 0; 
}

char *ctime(const time_t *tempo)

Converte um valor de time_t diretamente em uma string legível formatada (equivalente a asctime(localtime(tempo))).

Recebe: Um ponteiro para uma variável time_t.
Retorna: Uma string formatada como 'DiaSem Mês Dia Hora:Min:Seg Ano\n\0'.

Exemplo

#include <time.h>

struct tm *gmtime(const time_t *tempo)

Converte um valor de time_t numa estrutura struct tm com base no Meridiano de Greenwich (UTC/GMT).

Recebe: Um ponteiro para uma variável time_t.
Retorna: Um ponteiro para struct tm modificado para a hora universal, facilitando ajustes de fuso horário manualmente.

Exemplo

        
// EXEMPLO GMTIME()
#include <stdio.h>
#include <time.h>

#define BRT -3 
#define JST 9

int main() { 
    time_t tempo; 
    time(&tempo);

    struct tm *tempo0 = gmtime(&tempo);

    printf("Horário no Japão: %02d:%02d:%02d\n", 
           (tempo0->tm_hour + JST) % 24, tempo0->tm_min, tempo0->tm_sec);
           
    printf("Horário no Brasil: %02d:%02d:%02d\n", 
           (tempo0->tm_hour + BRT + 24) % 24, tempo0->tm_min, tempo0->tm_sec);

    return 0; 
}

struct tm *localtime(const time_t *tempo)

Converte um valor de time_t numa estrutura struct tm ajustada automaticamente para o fuso horário local do sistema.

Recebe: Um ponteiro para uma variável time_t.
Retorna: Um ponteiro para struct tm com as informações do tempo local.

Exemplo

        
// EXEMPLO LOCALTIME()
#include <stdio.h>
#include <time.h>

int main() { 
    time_t tempo; 
    time(&tempo); 

    struct tm *tempo0 = localtime(&tempo);

    printf("Horário atual: %02d:%02d:%02d\n", 
           tempo0->tm_hour, tempo0->tm_min, tempo0->tm_sec);

    return 0; 
}

size_t strftime(char *string_armazena_data, size_t tamanho, const char *formato, const struct tm *tempo)

Permite formatar as informações de tempo contidas em um struct tm e armazená-las em uma string com formatação customizada.

Recebe: Ponteiro de destino (string), limite de tamanho (bytes), string de formatação (ex: "%c", "%Y-%m-%d") e ponteiro struct tm de origem.
Retorna: A quantidade de caracteres armazenados na string (sem contar o null-byte final). Se exceder o tamanho, retorna 0.

Exemplo

        
// EXEMPLO STRFTIME()
#include <stdio.h>
#include <time.h>
#include <stdlib.h>

int main() { 
    time_t tempo; 
    time(&tempo);
    struct tm *tempo0 = localtime(&tempo);

    char *string_armazena_data = (char *)malloc(100 * sizeof(char));
    
    // %c representa data e hora padronizadas para o locale
    strftime(string_armazena_data, 100, "%c", tempo0);

    printf("%s\n", string_armazena_data);
    
    free(string_armazena_data);
    return 0; 
}
